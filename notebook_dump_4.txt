011: 
012:         # Create a figure and axes for the plot.
013:         fig_ridge, ax_ridge = plt.subplots(figsize=(12, 2 + len(sorted_df) * 0.7))
014: 
015:         # --- 1a. Create a color palette ---
016:         # Use a colormap to get a unique color for each variant.
017:         colors = plt.colormaps.get_cmap('viridis')(np.linspace(0.1, 0.9, len(sorted_df)))
018: 
019:         # --- 2. Dynamically Determine X-axis Range ---
020:         # Ensure all distributions and their credible intervals are fully visible.
021:         max_x = 0
022:         for i, row in sorted_df.iterrows():
023:             percentile_999 = stats.beta.ppf(0.999, row['posterior_alpha'], row['posterior_beta'])
024:             if percentile_999 > max_x:
025:                 max_x = percentile_999
026:         x = np.linspace(0, max_x * 1.05, 1000)
027: 
028:         # --- 3. Plot Each Variant as a Ridge ---
029:         y_offset_step = 0.8  # Controls vertical spacing between ridges
030: 
031:         for idx_color, (row, color) in enumerate(zip(sorted_df.itertuples(), colors)):
032:             y_offset = i * y_offset_step
033: 
034:             # Calculate the Probability Density Function (PDF)
035:             pdf = stats.beta.pdf(x, row.posterior_alpha, row.posterior_beta)
036: 
037:             # Plot the main distribution curve with a label for the legend.
038:             ax_ridge.plot(x, pdf + y_offset, color=color, lw=1.5, label=row.variant)
039: 
040:             # Add a light fill for the entire distribution
041:             ax_ridge.fill_between(x, y_offset, pdf + y_offset, alpha=0.2, color=color)
042: 
043:             # --- 4. Calculate and Shade the 95% Credible Interval ---
044:             # This interval contains the true conversion rate with 95% probability.
045:             ci_low, ci_high = stats.beta.ppf([0.025, 0.975], row.posterior_alpha, row.posterior_beta)
046: 
047:             # Create a mask for the x-values within the credible interval
048:             ci_mask = (x >= ci_low) & (x <= ci_high)
049: 
050:             # Add a darker shade on top for the 95% credible interval
051:             ax_ridge.fill_between(x[ci_mask], y_offset, pdf[ci_mask] + y_offset, alpha=0.4, color=color)

001: # --- 1. Simulation Setup ---
002: # Define the number of random samples to generate for each variant's distribution.
003: # A larger number of samples leads to more stable and accurate estimates of our metrics.
004: N_SAMPLES = 100000
005: 
006: # We will store the generated samples in a dictionary, with variant names as keys.
007: app.posterior_samples = {}
008: 
009: 
010: # --- 2. Run Simulation ---
011: # Check if the dataframe 'app.df' with posterior parameters exists to avoid errors.
012: if app.df is not None and 'posterior_alpha' in app.df.columns:
013: 
014:     # Iterate over each variant (row) in the DataFrame.
015:     for idx_row, row in app.df.iterrows():
016:         variant_name = row['variant']
017:         p_alpha = row['posterior_alpha']
018:         p_beta = row['posterior_beta']
019: 
020:         # Generate N_SAMPLES from the Beta distribution defined by the variant's
021:         # posterior parameters. Each sample represents a plausible "true"
022:         # conversion rate for that variant, according to our model.
023:         samples = stats.beta.rvs(a=p_alpha, b=p_beta, size=N_SAMPLES)
024: 
025:         # Store the resulting array of samples in our dictionary.
026:         app.posterior_samples[variant_name] = samples
027: 
028:     # --- 3. Output: Simulation Summary ---
029:     # The simulation is complete. The following is a summary of the process.
030:     print("--- Monte Carlo Simulation Summary ---")
031:     print(f"✅ Simulation completed successfully.")
032:     print(f"   - Samples generated per variant: {N_SAMPLES:,}")
033:     print(f"   - Variants simulated: {list(app.posterior_samples.keys())}")
034: 
035:     print("\nData Preview (first 3 samples for each variant):")
036:     for variant, samples in app.posterior_samples.items():
037:         preview = [round(float(s), 6) for s in list(samples)[:3]]
038:         print(f"  - {variant}: {preview}")
